async function getSignal() { const pair = document.getElementById('pair').value; const result = document.getElementById('result'); result.style.display = 'block'; result.className = 'signal'; result.innerHTML = 'Завантаження...'; try { const response = await fetch(`https://signal-ai-backend-production.up.railway.app/signal?pair=${pair}`); const data = await response.json(); if (data.error) { result.innerHTML = 'Помилка отримання даних'; return; } const isCall = data.direction === 'CALL'; const isPut = data.direction === 'PUT'; result.className = 'signal ' + (isCall ? 'call' : isPut ? 'put' : ''); result.innerHTML = ` ${data.pair}
${data.direction === 'WAIT' ? 'ЧЕКАТИ' : data.direction + (isCall ? ' ↑' : ' ↓')}
Впевненість: ${data.confidence}%
RSI: ${data.rsi} `; } catch (e) { result.innerHTML = 'Помилка з\'єднання'; } }
